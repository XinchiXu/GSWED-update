var updatedWaterColAfrica = ee.ImageCollection(imageCollection.toList(100))  //.filterBounds(geometry)
.map(function(img){
    img=img.where(img.eq(2),1);
    var bandNames=img.bandNames()//.slice(0,46);
    var NDVICols=ee.ImageCollection(bandNames.map(function(BandNameElement){
      var stringLength=ee.String(BandNameElement).length();
      var stryearBegin=ee.String(BandNameElement).slice(-10,-1)
      var startIndex=ee.String(BandNameElement).rindex(stryearBegin)
      var DateString=ee.String(BandNameElement).slice(startIndex,startIndex.add(10))
      var yearStr=ee.Number.parse(DateString.slice(0,4));
      var monthStr=ee.Number.parse(DateString.slice(5,7));
      var DayStr=ee.Number.parse(DateString.slice(8,10));
      return ee.Image(img.select([BandNameElement])).rename(['B1']).cast({"B1": "uint8"}, ["B1"])
    .set('system:time_start', ee.Date.fromYMD(yearStr.int(), monthStr.int(), DayStr.int()).millis())
    .set('bandName',BandNameElement)
    .set('year',yearStr)
    }))
   
    var list=ee.List([2023])
    var cols=list.map(function(year){
      var cols=NDVICols.filterMetadata('year','equals',year);
      var bandNumber=cols.toBands().bandNames().length();
      return cols.set('year',year).set('bandNumber',bandNumber);
    })
    
    var countCollection=cols.map(function(col){
      var count=ee.ImageCollection(col).map(function(image){
      return image.updateMask(image.eq(1));
    }).count();
      var year=ee.ImageCollection(col).get('year')
      var bandNumber=ee.Number(ee.ImageCollection(col).get('bandNumber'))
      return count.set("year",year)//.divide(bandNumber);
    })
    countCollection = ee.ImageCollection(countCollection)
    //////////////////////////////////////////////////
    var updatedWaterCol = cols.map(function(waterCollection){
      waterCollection=ee.ImageCollection(waterCollection)
      var updatedWater = waterCollection.map(function(water){
            /////////spatial Mask
          var waterMask=water.updateMask(water.eq(1))
          var conn=waterMask.connectedPixelCount({
            // maxSize:1024,
             maxSize:100,
             eightConnected:true
             }).reproject("EPSG:4326", null, 250)
          var spatialMask=conn.lte(5);
          var year = water.get("year");
          var temporalMask = countCollection.filter(ee.Filter.eq("year",year)).first().lte(5);
          var updatedWater=water.where(water.eq(1).and(temporalMask.eq(1)).and(spatialMask.eq(1)),4)
          return updatedWater.rename([water.get("bandName")])
      })
      return updatedWater.toBands()
    })
    return ee.ImageCollection(updatedWaterCol).toBands()
})
print("updatedWaterColAfrica",updatedWaterColAfrica.first())

function batchRename(image){
  var rename=image.bandNames().map(function(name){
    name = ee.String(name)
    name = ee.List(name.split("_"))
    return ee.String(name.get(2)).cat("_").cat(ee.String(name.get(3))).cat("_").cat(ee.String(name.get(4))).cat("_").cat(ee.String(name.get(5)))
  })
  return image.rename(rename);
}

//export images to Asset
function exportImageColAsset(imgCol) { 
  var indexList = imgCol.reduceColumns(ee.Reducer.toList(), ["system:index"]) 
                        .get("list"); 
  print(indexList)
  indexList.evaluate(function(indexs) { 
    for (var i=0; i<indexs.length; i++) { 
      var image = ee.Image(imgCol.filter(ee.Filter.eq("system:index", indexs[i])).first()); 
      //Map.addLayer(image.geometry())
      Export.image.toAsset({ 
        image: image, 
        description:ee.String(image.get("system:index")).getInfo(),
        scale: 250,
        region: image.geometry(),
        crs:"EPSG:4326",
        assetId: ee.String('users/qianrswaterr/GSWED_Africa2023_4326v2/').cat(ee.String(image.get("system:index"))).getInfo(),
        maxPixels: 1e13,
        pyramidingPolicy: {'.default': 'sample'}
      }); 
    } 
  }); 
}

        maxPixels: 1e13,

      });

  });
