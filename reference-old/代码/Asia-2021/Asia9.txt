function rmCloud(image) {
    var cloud0BitMask = (1 << 0);
    var cloud1BitMask = (1 << 1);
    var snowIceBitMask = (1 << 12);
    var qa = image.select("State");
    var mask = qa.bitwiseAnd(cloud0BitMask).eq(0)
                 .and(qa.bitwiseAnd(cloud1BitMask).eq(0))
    .and(qa.bitwiseAnd(snowIceBitMask).eq(0));
    return image.updateMask(land.eq(1)).updateMask(mask);
}

// set lag day range to use for scene interpolation (also retrieves images within this 
// time span on either side of desired composite range)
var lagRange = 24;
var NDVICols=ee.ImageCollection(bandNames.map(function(BandNameElement){
  var stringLength=ee.String(BandNameElement).length();
  var stryearBegin=ee.String(BandNameElement).slice(-10,-1)
  var startIndex=ee.String(BandNameElement).rindex(stryearBegin)
  var DateString=ee.String(BandNameElement).slice(startIndex,startIndex.add(10))
  var yearStr=ee.Number.parse(DateString.slice(0,4));
  var monthStr=ee.Number.parse(DateString.slice(5,7));
  var DayStr=ee.Number.parse(DateString.slice(8,10));
   
  return ee.Image(OriginalImage.select([BandNameElement])).rename(['B1']).cast({"B1": "int8"}, ["B1"])
.set('system:time_start', ee.Date.fromYMD(yearStr.int(), monthStr.int(), DayStr.int()).millis())
.set('bandName',BandNameElement)
}))
// Looks for all images up to 'lagRange' days away from the current image.
var maxDiffFilter = ee.Filter([
  ee.Filter.maxDifference({
    difference: lagRange * 24 * 60 * 60 * 1000,
    leftField: 'system:time_start',
    rightField: 'system:time_start'
  })]);
// Images after, sorted in descending order (so closest is last).
//here we cannot remove the equals, otherwise the timeseries will lost last element
var  FilterAfter= ee.Filter.and(maxDiffFilter, ee.Filter.lessThanOrEquals('system:time_start', null, 'system:time_start'))
var NDVI_AfterJoinedCols = ee.Join.saveAll('after', 'system:time_start', false).apply(NDVICols, NDVICols, FilterAfter)
// Images before, sorted in ascending order (so closest is last).
//here we cannot remove the equals, otherwise the timeseries will lost first element
var FilterBefore = ee.Filter.and(maxDiffFilter, ee.Filter.greaterThanOrEquals('system:time_start', null, 'system:time_start'))
var NDVI_Before_AfterJoinedCols = ee.Join.saveAll('before', 'system:time_start', true).apply(NDVI_AfterJoinedCols, NDVICols, FilterBefore)
var interpolation3 = ee.ImageCollection(NDVI_Before_AfterJoinedCols.map(function(img) {
  var image = ee.Image(img);
  var ImgbandName=image.get('bandName');
  var needFixMask=image.eq(ee.Image(6));
//because we need to exclusive the image itself
//for the begin
  var ImageListBefore=ee.List(image.get('before'));
  var CountBefore=ImageListBefore.length();
//operate on the before list
  var beforeCols = ee.Algorithms.If(CountBefore.gte(3),
  ee.ImageCollection.fromImages(ImageListBefore.slice(0,3)),
  ee.ImageCollection.fromImages(ImageListBefore.slice(0,CountBefore)));
//for the end
  var ImageListAfter=ee.List(image.get('after'));
  var CountAfter=ImageListAfter.length();
//operate on the after list
  var afterCols = ee.Algorithms.If(CountAfter.gte(3),
  ee.ImageCollection.fromImages(ImageListAfter.slice(0,3)),
  ee.ImageCollection.fromImages(ImageListAfter.slice(0,CountAfter)));
//combine the before and after cols  
  var combinedBeforeAfterCols=ee.ImageCollection(beforeCols).merge(afterCols);
  // the most common value at each pixel of six 
  var MostCommonImage= combinedBeforeAfterCols.map(function(image){
    return image.updateMask(image.neq(6));
  }).mode();
  MostCommonImage=MostCommonImage.unmask(6);
  //begin to interp
  var interp =needFixMask.not().multiply(image).add(needFixMask.multiply(MostCommonImage)).rename(['B1']);
  var mask=MostCommonImage.eq(1).and(image.neq(1)).and(needFixMask.eq(1));
  interp=interp.where(mask.eq(1),2);
  interp=interp.where(image.eq(2),2);
  //assgin the time to the processed var
  return interp.cast({"B1": "int8"}, ["B1"]).rename([ImgbandName])
  .set('system:time_start', img.get('system:time_start'))
  .set('bandName',ImgbandName).toUint8();
}))
interpolation3=batchRename(interpolation3.toBands());
return interpolation3.reproject('EPSG:4326',null,250);
})
